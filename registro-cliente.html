// app-registro-universal.js import { auth, db, googleProvider, signInWithPopup, createUserWithEmailAndPassword, doc, setDoc, serverTimestamp } from "./firebase.js"; const registroForm = document.getElementById("clienteForm"); const googleBtn = document.getElementById("loginGoogle"); const rol = registroForm.getAttribute("data-rol") || "CLIENTES"; // Define roles: CLIENTE o TECNICO // --- REGISTRO CON EMAIL Y CONTRASEÑA --- registroForm.addEventListener("submit", async (e) => { e.preventDefault(); const submitBtn = document.getElementById("submitBtn"); const formData = new FormData(registroForm); const email = formData.get("correo"); const password = "FixGo" + (formData.get("telefono") || '').slice(-4); // Contraseña temporal toggleButtonState(submitBtn, true, "CREANDO CUENTA..."); try { const userCredential = await createUserWithEmailAndPassword(auth, email, password); const user = userCredential.user; await guardarEnFirestore(user, { nombre: formData.get("nombre"), telefono: formData.get("telefono"), direccion: formData.get("direccion"), correo: email, rol: rol, uid: user.uid }); alert("¡Cuenta creada con éxito! Tu contraseña temporal es: " + password); window.location.href = "index.html"; } catch (error) { console.error("Error en registro:", error); alert("Error: El correo ya existe o los datos son inválidos."); } finally { toggleButtonState(submitBtn, false, "CREAR MI CUENTA"); } }); // --- REGISTRO CON GOOGLE --- googleBtn.addEventListener("click", async () => { try { const result = await signInWithPopup(auth, googleProvider); const user = result.user; await guardarEnFirestore(user, { nombre: user.displayName, correo: user.email, rol: rol, uid: user.uid, registroSocial: true }); window.location.href = "index.html"; } catch (error) { console.error("Error Google:", error); } }); // Función auxiliar para guardar datos en Firestore async function guardarEnFirestore(user, datos) { const coleccion = (datos.rol === "CLIENTES") ? "clientes" : "tecnicos"; await setDoc(doc(db, coleccion, user.uid), { ...datos, fechaRegistro: serverTimestamp() }); } // Función para alternar el estado del botón function toggleButtonState(button, disable, text) { button.innerText = text; button.disabled = disable; }
