// app-registro-universal.js import { auth, db, googleProvider, signInWithPopup, createUserWithEmailAndPassword } from "./firebase-auth.js"; import { doc, setDoc, serverTimestamp } from "https://www.gstatic.com/firebasejs/10.8.0/firebase-firestore.js"; const registroForm = document.getElementById("clienteForm"); const googleBtn = document.getElementById("loginGoogle"); const rol = registroForm.getAttribute("data-rol") || "CLIENTES"; // Detecta si es CLIENTE o TECNICO // --- REGISTRO CON EMAIL Y CONTRASEÑA --- registroForm.addEventListener("submit", async (e) => { e.preventDefault(); const btn = document.getElementById("submitBtn"); const formData = new FormData(registroForm); // Generamos una contraseña genérica o podrías añadir un campo password al HTML // Por simplicidad en el registro rápido, usaremos el teléfono como pass temporal // o puedes añadir el input en el HTML. Aquí asumiremos que ya existe o lo creamos. const email = formData.get("correo"); const password = "FixGo" + formData.get("telefono").slice(-4); // Password temporal segura btn.innerText = "CREANDO CUENTA..."; btn.disabled = true; try { const userCredential = await createUserWithEmailAndPassword(auth, email, password); const user = userCredential.user; await guardarEnFirestore(user, { nombre: formData.get("nombre"), telefono: formData.get("telefono"), direccion: formData.get("direccion"), correo: email, rol: rol, uid: user.uid }); alert("¡Cuenta creada con éxito! Tu contraseña temporal es: " + password); window.location.href = "index.html"; } catch (error) { console.error("Error en registro:", error); alert("Error: El correo ya existe o los datos son inválidos."); } finally { btn.innerText = "CREAR MI CUENTA"; btn.disabled = false; } }); // --- REGISTRO CON GOOGLE --- googleBtn.addEventListener("click", async () => { try { const result = await signInWithPopup(auth, googleProvider); const user = result.user; // Para Google, como no tenemos formulario, guardamos datos básicos await guardarEnFirestore(user, { nombre: user.displayName, correo: user.email, rol: rol, uid: user.uid, registroSocial: true }); window.location.href = "index.html"; } catch (error) { console.error("Error Google:", error); } }); // Función auxiliar para centralizar el guardado async function guardarEnFirestore(user, datos) { const coleccion = datos.rol === "CLIENTE" ? "clientes" : "tecnicos"; await setDoc(doc(db, coleccion, user.uid), { ...datos, fechaRegistro: serverTimestamp() }); }
