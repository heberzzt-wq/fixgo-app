function b2cStorageSignedIn() {
  return request.auth != null;
}

// Replica la autoridad maestra del ruleset Firestore real. No confía en un campo rol
// editable dentro de users/{uid}.
function b2cStorageIsAdmin() {
  return b2cStorageSignedIn() && (
    request.auth.token.email == "hebertoh-m@hotmail.com" ||
    request.auth.uid == "nNhwy3Mx4pTvc8TZVh1tyTMFwhC2"
  );
}

function b2cStorageService(serviceId) {
  return firestore.get(
    /databases/(default)/documents/services/$(serviceId)
  ).data;
}

function b2cStorageIsAssignedTechnician(serviceId) {
  return b2cStorageSignedIn() &&
    b2cStorageService(serviceId).tecnico_id == request.auth.uid;
}

function b2cStorageIsOwningCustomer(serviceId) {
  return b2cStorageSignedIn() &&
    b2cStorageService(serviceId).cliente_id == request.auth.uid;
}

function b2cStorageImageType() {
  return request.resource.contentType.matches("image/(jpeg|png|webp)");
}

function b2cStorageVideoType() {
  return request.resource.contentType.matches("video/(webm|mp4|quicktime)");
}

function b2cStoragePhotoEvent(eventType) {
  return eventType in [
    "arrival",
    "arrival_fallback",
    "diagnostic_before",
    "work_before",
    "work_after",
    "customer_no_show",
    "customer_denied_access",
    "service_completion"
  ];
}

function b2cStorageVideoEvent(eventType) {
  return eventType in [
    "customer_no_show_video",
    "customer_denied_access_video"
  ];
}

function b2cStorageMediaMatchesEvent(eventType) {
  return (
    b2cStoragePhotoEvent(eventType) &&
    b2cStorageImageType() &&
    request.resource.size <= 2 * 1024 * 1024
  ) || (
    b2cStorageVideoEvent(eventType) &&
    b2cStorageVideoType() &&
    request.resource.size <= 6 * 1024 * 1024
  );
}

match /b2c_evidence/{serviceId}/{actorUid}/{eventType}/{fileName} {
  allow read: if b2cStorageIsAdmin() ||
    b2cStorageIsAssignedTechnician(serviceId) ||
    b2cStorageIsOwningCustomer(serviceId);

  allow create: if b2cStorageIsAssignedTechnician(serviceId) &&
    actorUid == request.auth.uid &&
    request.resource.size > 0 &&
    b2cStorageMediaMatchesEvent(eventType) &&
    request.resource.metadata.actorUid == request.auth.uid &&
    request.resource.metadata.actorRole == "tecnico" &&
    request.resource.metadata.serviceId == serviceId &&
    request.resource.metadata.eventType == eventType;

  allow update: if false;
  allow delete: if b2cStorageIsAdmin();
}

match /b2c_customer_evidence/{serviceId}/{customerId}/arrival_dispute/{fileName} {
  allow read: if b2cStorageIsAdmin() ||
    b2cStorageIsAssignedTechnician(serviceId) ||
    b2cStorageIsOwningCustomer(serviceId);

  allow create: if b2cStorageIsOwningCustomer(serviceId) &&
    customerId == request.auth.uid &&
    request.resource.size > 0 &&
    request.resource.size <= 2 * 1024 * 1024 &&
    b2cStorageImageType() &&
    request.resource.metadata.actorUid == request.auth.uid &&
    request.resource.metadata.actorRole == "cliente" &&
    request.resource.metadata.serviceId == serviceId &&
    request.resource.metadata.eventType == "customer_arrival_dispute" &&
    request.resource.metadata.dedupScope == "service_only" &&
    request.resource.metadata.dedupBackendPending == "true";

  allow update: if false;
  allow delete: if b2cStorageIsAdmin();
}

// Única apertura B2C dentro de la ruta legacy: firma PNG económica.
match /servicios/{serviceId}/{fileName} {
  allow read: if b2cStorageIsAdmin() ||
    b2cStorageIsAssignedTechnician(serviceId) ||
    b2cStorageIsOwningCustomer(serviceId);

  allow create: if b2cStorageIsAssignedTechnician(serviceId) &&
    fileName.matches("customer_signature_.*\\.png") &&
    request.resource.size > 0 &&
    request.resource.size <= 512 * 1024 &&
    request.resource.contentType == "image/png" &&
    request.resource.metadata.serviceId == serviceId &&
    request.resource.metadata.actorUid == request.auth.uid &&
    request.resource.metadata.actorRole == "tecnico" &&
    request.resource.metadata.eventType == "customer_signature" &&
    request.resource.metadata.base64Persisted == "false";

  // Las fotos legacy de este prefijo deben conservarse en el ruleset completo auditado.
  allow update: if false;
  allow delete: if b2cStorageIsAdmin();
}
