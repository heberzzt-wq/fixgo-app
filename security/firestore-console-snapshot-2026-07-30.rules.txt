service cloud.firestore {
  match /databases/{database}/documents {

    function isAuth() {
      return request.auth != null;
    }

    function isAdmin() {
      return isAuth() && (
        request.auth.token.get('admin', false) == true ||
        request.auth.token.get('email', '') == 'hebertoh-m@hotmail.com' ||
        request.auth.uid == 'nNhwy3Mx4pTvc8TZVh1tyTMFwhC2' ||
        (
          exists(/databases/$(database)/documents/users/$(request.auth.uid)) &&
          get(/databases/$(database)/documents/users/$(request.auth.uid)).data.get('rol', '') in ['admin', 'ceo']
        )
      );
    }

    function userDoc() {
      return isAuth()
        ? get(/databases/$(database)/documents/users/$(request.auth.uid))
        : null;
    }

    function userData() {
      let doc = userDoc();
      return doc != null && doc.data != null ? doc.data : null;
    }

    function userRole() {
      let data = userData();
      return data != null ? data.rol : null;
    }

    function userTipo() {
      let data = userData();
      return data != null ? data.tipo_cuenta : null;
    }

    function userStatus() {
      let data = userData();
      return data != null ? data.status : null;
    }

    function isTecnico() {
      let role = userRole();
      return role != null &&
             role in ['tecnico', 'tecnico_gp', 'tecnico_interno'] &&
             userStatus() == 'activo';
    }

    function isDocumentReference(value) {
      return (value is string && value.size() > 0) ||
        (value is map && (
          (value.get('url', '') is string && value.get('url', '').size() > 0) ||
          (value.get('storage_path', '') is string && value.get('storage_path', '').size() > 0)
        ));
    }

    function technicianKycIsComplete(data) {
      let vehicle = data.get('vehiculo', {});
      let documents = data.get('documentos', {});
      let banking = data.get('datos_bancarios', {});
      let pedestrian = vehicle.get('tipo', '') == 'peaton';
      return isDocumentReference(data.get('foto_perfil', null)) &&
        isDocumentReference(documents.get('ine', null)) &&
        isDocumentReference(documents.get('csf', null)) &&
        banking.get('banco', '') is string && banking.get('banco', '').size() > 0 &&
        banking.get('clabe', '') is string && banking.get('clabe', '').matches('^[0-9]{18}$') &&
        vehicle.get('tipo', '') is string && vehicle.get('tipo', '').size() > 0 &&
        (pedestrian || (
          vehicle.get('placas', '') is string && vehicle.get('placas', '').size() > 0 &&
          isDocumentReference(documents.get('licencia', null))
        ));
    }

    function isOperationalTechnician() {
      let data = userData();
      return data != null &&
        data.rol in ['tecnico', 'tecnico_gp', 'tecnico_interno'] &&
        data.estado == 'activo' &&
        data.status == 'activo' &&
        data.get('kyc', {}).get('aprobado', false) == true &&
        data.get('suspendido', false) != true &&
        data.get('disponible', false) == true &&
        technicianKycIsComplete(data);
    }

    function technicianInitialStateIsSafe(data) {
      return data.rol == 'tecnico' &&
        data.estado in ['registro_iniciado', 'documentos_pendientes'] &&
        data.status == data.estado &&
        data.disponible == false &&
        data.kyc.aprobado == false;
    }

    function userAuthorityFieldsUnchanged() {
      return request.resource.data.rol == resource.data.rol &&
        request.resource.data.tipo_cuenta == resource.data.tipo_cuenta &&
        (
          (
            request.resource.data.estado == resource.data.estado &&
            request.resource.data.status == resource.data.status
          ) ||
          (
            resource.data.rol == 'tecnico' &&
            resource.data.estado in ['registro_iniciado', 'documentos_pendientes', 'rechazado'] &&
            request.resource.data.estado == 'pendiente_revision' &&
            request.resource.data.status == 'pendiente_revision' &&
            request.resource.data.disponible == false
          )
        ) &&
        request.resource.data.get('suspendido', false) == resource.data.get('suspendido', false) &&
        request.resource.data.get('kyc', {}).get('aprobado', false) == resource.data.get('kyc', {}).get('aprobado', false) &&
        request.resource.data.get('kyc', {}).get('aprobado_por', null) == resource.data.get('kyc', {}).get('aprobado_por', null) &&
        request.resource.data.get('kyc', {}).get('aprobado_at', null) == resource.data.get('kyc', {}).get('aprobado_at', null);
    }

    function serviceFinancialFieldsUnchanged() {
      return !request.resource.data.diff(resource.data).affectedKeys().hasAny([
        'retencion_inicial', 'costo_final', 'monto_total', 'pago_estado',
        'stripe_session_id', 'stripe_payment_intent', 'saldo_pagado',
        'ledger_transaction_id', 'liquidacion_estado', 'fecha_liquidacion'
      ]);
    }

    function customerServiceUpdateIsSafe() {
      let before = resource.data.estado;
      let after = request.resource.data.estado;
      return (
          after == before ||
          (before == 'iniciado_stripe' && after == 'cancelado')
        ) && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          'estado', 'cancelado_razon', 'cancelado_at', 'actualizado_at',
          'llegada_notificacion_estado', 'llegada_notificacion_mostrada_at',
          'llegada_notificacion_version', 'llegada_espera_segundos',
          'llegada_cliente_respuesta', 'llegada_cliente_respuesta_at',
          'llegada_revision_requerida', 'llegada_resolucion_automatica_bloqueada',
          'llegada_disputa_cliente'
        ]);
    }

    function technicianServiceTransitionIsSafe() {
      let before = resource.data.estado;
      let after = request.resource.data.estado;
      let same = after == before;
      let allowed =
        (before == 'asignado' && after == 'en_camino') ||
        (before == 'en_camino' && after == 'en_sitio') ||
        (
          before == 'trabajando' &&
          after == 'finalizado' &&
          request.resource.data.get('cierre_operativo_completado', false) == true &&
          request.resource.data.get('cierre_financiero_pendiente_backend', false) == true &&
          request.resource.data.get('cierre_legacy_financiero_ejecutado', true) == false &&
          request.resource.data.get('work_evidence_binding_path', null) is string
        );
      return (same || allowed) &&
        !request.resource.data.diff(resource.data).affectedKeys().hasAny([
          'cliente_id', 'cliente_nombre', 'cliente_telefono', 'metodo_pago',
          'categoria', 'categoria_id', 'sub_servicio', 'tipo',
          'tecnico_id', 'tecnico_nombre', 'tecnico_nombre_fiscal',
          'tecnico_rfc', 'tecnico_logo_factura', 'tecnico_vehiculo', 'tecnico_placas'
        ]);
    }

    function tienePaseCaseta() {
      let data = userData();
      return data != null &&
             data.tecnico_placas != null &&
             data.tecnico_placas != "" &&
             data.tecnico_placas != "000-000";
    }

    function userTenant() {
      let data = userData();
      return data != null ? (data.edificioId != null ? data.edificioId : data.tenantId) : null;
    }

    function b2cServiceExists(serviceId) {
      return exists(/databases/$(database)/documents/services/$(serviceId));
    }

    function b2cService(serviceId) {
      return get(/databases/$(database)/documents/services/$(serviceId)).data;
    }

    function b2cIsAssignedTechnician(serviceId) {
      return isAuth() &&
        b2cServiceExists(serviceId) &&
        b2cService(serviceId).get('tecnico_id', null) == request.auth.uid;
    }

    function b2cIsOwningCustomer(serviceId) {
      return isAuth() &&
        b2cServiceExists(serviceId) &&
        b2cService(serviceId).get('cliente_id', null) == request.auth.uid;
    }

    function b2cIsServiceParty(serviceId) {
      return isAdmin() || b2cIsAssignedTechnician(serviceId) || b2cIsOwningCustomer(serviceId);
    }

    function b2cEventServiceId(data) {
      return 'serviceId' in data ? data.serviceId : data.service_id;
    }

    function b2cEventType(data) {
      return 'eventType' in data ? data.eventType : data.event_type;
    }

    function b2cEventActorUid(data) {
      return 'actor' in data ? data.actor.uid : data.actor_uid;
    }

    function b2cEventActorRole(data) {
      return 'actor' in data ? data.actor.role : data.actor_role;
    }

    function b2cEvidenceActorValid(serviceId, data) {
      let actorUid = b2cEventActorUid(data);
      let actorRole = b2cEventActorRole(data);
      return actorUid == request.auth.uid && (
        (actorRole == 'tecnico' && b2cIsAssignedTechnician(serviceId)) ||
        (actorRole == 'cliente' && b2cIsOwningCustomer(serviceId))
      );
    }

    function b2cEvidenceEventAllowed(eventType) {
      return eventType in [
        'arrival', 'arrival_fallback', 'diagnostic_before', 'work_before', 'work_after',
        'customer_no_show', 'customer_denied_access', 'customer_no_show_video',
        'customer_denied_access_video', 'service_completion', 'customer_arrival_dispute'
      ];
    }

    function b2cTechnicianConsentValid(serviceId, expectedEvent) {
      return b2cIsAssignedTechnician(serviceId) &&
        request.resource.data.actor_uid == request.auth.uid &&
        request.resource.data.actor_role == 'tecnico' &&
        request.resource.data.event_type == expectedEvent;
    }

    function b2cStoredSignatureValid(serviceId, technicianId, signature) {
      return signature is map &&
        signature.get('present', false) == true &&
        signature.get('sha256', '').matches('^[a-f0-9]{64}$') &&
        signature.get('storage_path', '').matches('^servicios/' + serviceId + '/customer_signature_.*\\.png$') &&
        signature.get('download_url', '') is string &&
        signature.get('download_url', '').size() > 0 &&
        signature.get('base64_persisted', true) == false;
    }

    function b2cCompleteBindingValid(serviceId, technicianId, data) {
      return data.get('service_id', '') == serviceId &&
        data.get('technician_id', '') == technicianId &&
        data.get('before', {}).get('sha256', '').matches('^[a-f0-9]{64}$') &&
        data.get('before', {}).get('storage_path', '').matches('^b2c_evidence/' + serviceId + '/' + technicianId + '/work_before/.*') &&
        data.get('before', {}).get('download_url', '') is string &&
        data.get('before', {}).get('download_url', '').size() > 0 &&
        data.get('after', {}).get('sha256', '').matches('^[a-f0-9]{64}$') &&
        data.get('after', {}).get('storage_path', '').matches('^b2c_evidence/' + serviceId + '/' + technicianId + '/work_after/.*') &&
        data.get('after', {}).get('download_url', '') is string &&
        data.get('after', {}).get('download_url', '').size() > 0 &&
        b2cStoredSignatureValid(serviceId, technicianId, data.get('signature', {}));
    }

      allow create: if isAuth() && (
        isAdmin() ||
        userRole() in [
          'admin_b2b', 'ceo', 'admin', 'asistente_admin', 'recepcion',
          'seguridad', 'seguridad_interna', 'seguridad_24_7'
        ]
      );
      allow read: if isAuth();
      allow update, delete: if isAdmin();
    }

    match /anuncios_b2b/{anuncioId} {
      allow read: if isAuth() && (
        isAdmin() ||
        userData().edificioId == resource.data.edificioId ||
        userRole() in ['admin_b2b', 'ceo', 'admin', 'asistente_admin', 'recepcion', 'seguridad', 'seguridad_interna', 'seguridad_24_7']
      );
      allow create: if isAuth() && (
        isAdmin() ||
        userRole() in ['admin_b2b', 'ceo', 'admin', 'asistente_admin', 'recepcion', 'seguridad', 'seguridad_interna', 'seguridad_24_7']
      );
      allow update, delete: if isAdmin() || (isAuth() && request.auth.uid == resource.data.autor_uid);
    }

    match /alertas_seguridad/{document} {
      allow read, write: if isAuth();
    }

    match /servicios_b2b/{servicioId} {
      allow read: if isAuth() && (
        isAdmin() || isTecnico() || userRole() == 'admin_b2b' || userTipo() == 'B2B'
      );
      allow create: if isAuth() && (isAdmin() || isTecnico() || userTipo() == 'B2B');
      allow update: if isAuth() && (
        isAdmin() ||
        (isTecnico() && userData().edificioId == resource.data.edificioId && tienePaseCaseta()) ||
        (userTipo() == 'B2B' && resource.data.edificioId == userData().edificioId)
      );
    }

    match /config_rutinas/{edificioId} {
      allow read: if isAuth() && (isTecnico() || isAdmin() || userTipo() == 'B2B');
      allow write: if isAdmin() || (isAuth() && userTipo() == 'B2B');
    }

    match /log_rutinas/{logId} {
      allow read: if isAuth() && (isTecnico() || isAdmin() || userTipo() == 'B2B');
      allow create: if isAuth() && isTecnico() && request.resource.data.tecnicoId == request.auth.uid && tienePaseCaseta();
    }

    match /bitacora_edificios/{bitId} {
      allow read, create: if isAuth() && (isTecnico() || isAdmin() || userTipo() == 'B2B');
    }

    match /transacciones/{txId} {
      allow read: if isAuth() && (resource.data.tecnico_id == request.auth.uid || isAdmin());
      allow create, update, delete: if false;
    }

    match /retiros/{retiroId} {
      allow read: if isAuth() && (resource.data.tecnico_id == request.auth.uid || isAdmin());
      allow create, update, delete: if false;
    }

    match /rastreo/{userId} {
      allow read: if isAuth();
      allow write: if isAuth() && (request.auth.uid == userId || isAdmin());
    }

    match /empresas_b2b/{empresaId} {
      allow read: if isAuth();

      match /areas/{areaId} {
        allow read, write: if isAuth();
      }

      match /activos/{activoId} {
        allow read: if isAuth();
        allow create, update: if isAuth() && (
          isAdmin() || (userRole() == 'admin_b2b' && userData().edificioId == empresaId)
        ) &&
        (!('especificaciones_tecnicas' in request.resource.data) || request.resource.data.especificaciones_tecnicas is map) &&
        (!('especificaciones_tecnicas' in request.resource.data) || !('potencia_hp' in request.resource.data.especificaciones_tecnicas) || request.resource.data.especificaciones_tecnicas.potencia_hp is number);

        match /bitacora/{bitId} {
          allow create, read: if isAuth();
        }
      }

      match /mantenimiento/{tareaId} {
        allow read: if isAuth();
        allow create: if isAuth() && (
          isAdmin() || (userRole() == 'admin_b2b' && userData().edificioId == empresaId)
        ) && request.resource.data.keys().hasAll(['id_tarea', 'id_activo', 'sistema', 'frecuencia', 'estado']) &&
        request.resource.data.estado in ['pendiente', 'en_proceso', 'completado'] &&
        !('potencia_hp' in request.resource.data) && !('voltaje' in request.resource.data);
        allow update: if isAuth() && (
          isAdmin() ||
          (userRole() == 'admin_b2b' && userData().edificioId == empresaId) ||
          (isTecnico() && userData().edificioId == empresaId && tienePaseCaseta())
        ) &&
        (!('estado' in request.resource.data) || request.resource.data.estado in ['pendiente', 'en_proceso', 'completado']) &&
        !('potencia_hp' in request.resource.data);
      }
    }

    match /gestia_system_modules/{moduloId} {
      allow read: if isAuth();
      allow write: if isAdmin();
    }

    match /gestia_records/{tenantId} {
      allow read: if true;
      match /{moduloId} {
        allow read: if true;
        match /{docId} {
          allow read, write: if true;
        }
      }
    }

    match /condominios/{condoId} {
      function accesoCondominio() {
        let edificio = userTenant();
        return edificio != null && edificio == condoId;
      }
      allow read: if isAuth() && (isAdmin() || accesoCondominio());
      allow write: if isAdmin();
    }

    match /configuracion/{docId} {
      allow read: if isAuth();
      allow write: if isAdmin();
    }

    match /flotilla_b2b/{tenantId}/{document=**} {
      allow read, write: if isAuth();
    }

    match /packages/{tenantId}/items/{docId} {
      allow read, write: if true;
    }

    match /packages/{tenantId}/{docId} {
      allow read, write: if true;
    }
  }
