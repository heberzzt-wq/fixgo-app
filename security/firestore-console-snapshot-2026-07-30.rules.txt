// FUENTE AUTORITATIVA VERSIONADA. No desplegar sin ejecutar las pruebas de reglas.
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

   // ======================================================
    // 👑 FUNCIONES MAESTRAS (NÚCLEO DE SEGURIDAD)
    // ======================================================

    function isAuth() {
      return request.auth != null;
    }

    // MODO DIOS: Reforzado con UID para asegurar soberanía de Heberto
    function isAdmin() {
      return isAuth() && (
        request.auth.token.get('admin', false) == true ||
        request.auth.token.get('email', '') == 'hebertoh-m@hotmail.com' ||
        request.auth.uid == 'nNhwy3Mx4pTvc8TZVh1tyTMFwhC2' ||
        (
          exists(/databases/$(database)/documents/users/$(request.auth.uid)) &&
          get(/databases/$(database)/documents/users/$(request.auth.uid)).data.get('rol', '') in ['admin', 'ceo']
        )
      );
    }

    // Lectura segura del documento de usuario
    function userDoc() {
      return isAuth() 
        ? get(/databases/$(database)/documents/users/$(request.auth.uid)) 
        : null;
    }

    // Datos del usuario con protección contra NULL (Cura el Loop de Login)
    function userData() {
      let doc = userDoc();
      return doc != null && doc.data != null ? doc.data : null;
    }

    // Getters específicos (Evitan errores de 'null.field')
    function userRole() {
      let data = userData();
      return data != null ? data.rol : null;
    }

    function userTipo() {
      let data = userData();
      return data != null ? data.tipo_cuenta : null;
    }

    function userStatus() {
      let data = userData();
      return data != null ? data.status : null;
    }

    function isTecnico() {
      let role = userRole();
      return role != null && 
             role in ['tecnico', 'tecnico_gp', 'tecnico_interno'] && 
             userStatus() == 'activo';
    }

    // Adapter Firestore de b2c-platform-contract-v1. CI comprueba paridad con la
    // autoridad neutral de browser/Functions.
    function isDocumentReference(value) {
      return (value is string && value.size() > 0) ||
        (value is map && (
          (value.get('url', '') is string && value.get('url', '').size() > 0) ||
          (value.get('storage_path', '') is string && value.get('storage_path', '').size() > 0)
        ));
    }

    function technicianKycIsComplete(data) {
      let vehicle = data.get('vehiculo', {});
      let documents = data.get('documentos', {});
      let banking = data.get('datos_bancarios', {});
      let pedestrian = vehicle.get('tipo', '') == 'peaton';
      return isDocumentReference(data.get('foto_perfil', null)) &&
        isDocumentReference(documents.get('ine', null)) &&
        isDocumentReference(documents.get('csf', null)) &&
        banking.get('banco', '') is string && banking.get('banco', '').size() > 0 &&
        banking.get('clabe', '') is string && banking.get('clabe', '').matches('^[0-9]{18}$') &&
        vehicle.get('tipo', '') is string && vehicle.get('tipo', '').size() > 0 &&
        (pedestrian || (
          vehicle.get('placas', '') is string && vehicle.get('placas', '').size() > 0 &&
          isDocumentReference(documents.get('licencia', null))
        ));
    }

    function isOperationalTechnician() {
      let data = userData();
      return data != null &&
        data.rol in ['tecnico', 'tecnico_gp', 'tecnico_interno'] &&
        data.estado == 'activo' &&
        data.status == 'activo' &&
        data.get('kyc', {}).get('aprobado', false) == true &&
        data.get('suspendido', false) != true &&
        data.get('disponible', false) == true &&
        technicianKycIsComplete(data);
    }

    function technicianInitialStateIsSafe(data) {
      return data.rol == 'tecnico' &&
        data.estado in ['registro_iniciado', 'documentos_pendientes'] &&
        data.status == data.estado &&
        data.disponible == false &&
        data.kyc.aprobado == false;
    }

    function userAuthorityFieldsUnchanged() {
      return request.resource.data.rol == resource.data.rol &&
        request.resource.data.tipo_cuenta == resource.data.tipo_cuenta &&
        (
          (
            request.resource.data.estado == resource.data.estado &&
            request.resource.data.status == resource.data.status
          ) ||
          (
            resource.data.rol == 'tecnico' &&
            resource.data.estado in ['registro_iniciado', 'documentos_pendientes', 'rechazado'] &&
            request.resource.data.estado == 'pendiente_revision' &&
            request.resource.data.status == 'pendiente_revision' &&
            request.resource.data.disponible == false
          )
        ) &&
        request.resource.data.get('suspendido', false) == resource.data.get('suspendido', false) &&
        request.resource.data.get('kyc', {}).get('aprobado', false) == resource.data.get('kyc', {}).get('aprobado', false) &&
        request.resource.data.get('kyc', {}).get('aprobado_por', null) == resource.data.get('kyc', {}).get('aprobado_por', null) &&
        request.resource.data.get('kyc', {}).get('aprobado_at', null) == resource.data.get('kyc', {}).get('aprobado_at', null);
    }

    function serviceFinancialFieldsUnchanged() {
      return !request.resource.data.diff(resource.data).affectedKeys().hasAny([
        'retencion_inicial', 'costo_final', 'monto_total', 'pago_estado',
        'stripe_session_id', 'stripe_payment_intent', 'saldo_pagado',
        'ledger_transaction_id', 'liquidacion_estado', 'fecha_liquidacion'
      ]);
    }

    function customerServiceUpdateIsSafe() {
      let before = resource.data.estado;
      let after = request.resource.data.estado;
      return (
          after == before ||
          (before == 'iniciado_stripe' && after == 'cancelado')
        ) && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          'estado', 'cancelado_razon', 'cancelado_at', 'actualizado_at',
          'llegada_notificacion_estado', 'llegada_notificacion_mostrada_at',
          'llegada_notificacion_version', 'llegada_espera_segundos',
          'llegada_cliente_respuesta', 'llegada_cliente_respuesta_at',
          'llegada_revision_requerida', 'llegada_resolucion_automatica_bloqueada',
          'llegada_disputa_cliente'
        ]);
    }

    function technicianServiceTransitionIsSafe() {
      let before = resource.data.estado;
      let after = request.resource.data.estado;
      let same = after == before;
      let allowed =
        (before == 'asignado' && after == 'en_camino') ||
        (before == 'en_camino' && after == 'en_sitio') ||
        (
          before == 'trabajando' &&
          after == 'finalizado' &&
          request.resource.data.get('cierre_operativo_completado', false) == true &&
          request.resource.data.get('cierre_financiero_pendiente_backend', false) == true &&
          request.resource.data.get('cierre_legacy_financiero_ejecutado', true) == false &&
          request.resource.data.get('work_evidence_binding_path', null) is string
        );
      return (same || allowed) &&
        !request.resource.data.diff(resource.data).affectedKeys().hasAny([
          'cliente_id', 'cliente_nombre', 'cliente_telefono', 'metodo_pago',
          'categoria', 'categoria_id', 'sub_servicio', 'tipo',
          'tecnico_id', 'tecnico_nombre', 'tecnico_nombre_fiscal',
          'tecnico_rfc', 'tecnico_logo_factura', 'tecnico_vehiculo', 'tecnico_placas'
        ]);
    }

    // 🔒 REGLA 2: EL CANDADO JONATHAN (Placas)
    function tienePaseCaseta() {
      let data = userData();
      return data != null && 
             data.tecnico_placas != null && 
             data.tecnico_placas != "" && 
             data.tecnico_placas != "000-000";
    }

    // 🧠 MOTOR NO-CODE: Helper local para normalizar acceso (By-pass de campos)
    function userTenant() {
      let data = userData();
      return data != null ? (data.edificioId != null ? data.edificioId : data.tenantId) : null;
    }

    // ======================================================
    // 🔑 ACCESO GLOBAL ADMIN (Blindaje Heberto)
    // ======================================================
    match /{document=**} {
      // Solo tú tienes permiso de ver TODO el mapa de la base de datos
      allow read, write: if request.auth != null && request.auth.uid == 'nNhwy3Mx4pTvc8TZVh1tyTMFwhC2';
    }

    // ======================================================
    // 🔑 VALIDADOR DE LLAVES MAESTRAS (B2B KEYS)
    // ======================================================
    match /b2b_keys/{keyId} {
      allow read: if true; 
      allow write: if isAdmin();
    }

    // ======================================================
    // 👤 GESTIÓN DE USUARIOS (SSoT) - AISLAMIENTO B2B
    // ======================================================
    match /users/{userId} {
      allow read: if isAuth() && (
        request.auth.uid == userId ||
        isAdmin() ||
        (
          userRole() == 'admin_b2b' &&
          resource.data.edificioId == userData().edificioId
        )
      );

      allow create: if isAuth() && (
          (
            request.auth.uid == userId &&
            request.resource.data.uid == userId &&
            (
              technicianInitialStateIsSafe(request.resource.data) ||
              (
                request.resource.data.rol == 'cliente' &&
                request.resource.data.estado == 'activo' &&
                request.resource.data.status == 'activo' &&
                request.resource.data.tipo_cuenta == 'B2C' &&
                request.resource.data.get('pagos', {}).get('stripe_autorizado', true) == false &&
                request.resource.data.get('pagos', {}).get('efectivo_autorizado', true) == false &&
                request.resource.data.get('efectivo_autorizado', false) == false
              ) ||
              (
                request.resource.data.rol == 'admin_b2b' &&
                request.resource.data.tipo_cuenta == 'B2B' &&
                request.resource.data.estado == 'activo' &&
                request.resource.data.status == 'activo' &&
                request.resource.data.b2b_key_id is string &&
                exists(/databases/$(database)/documents/b2b_keys/$(request.resource.data.b2b_key_id)) &&
                get(/databases/$(database)/documents/b2b_keys/$(request.resource.data.b2b_key_id)).data.edificioId == request.resource.data.edificioId
              )
            )
          ) ||
          isAdmin() ||
          (userRole() == "admin_b2b" && request.resource.data.edificioId == userData().edificioId)
      );

      allow update: if isAuth() && (
          isAdmin() ||
          (
            request.auth.uid == userId &&
            userAuthorityFieldsUnchanged() &&
            !request.resource.data.diff(resource.data).affectedKeys().hasAny([
              'rol', 'tipo_cuenta', 'suspendido', 'verificado',
              'nivel', 'reputacion', 'servicios_completados', 'wallet', 'currency',
              'comision', 'aprobadoEn', 'pagos', 'efectivo_autorizado',
              'vehiculo_tipo', 'placas', 'logistica'
            ]) &&
            !request.resource.data.get('documentos', {}).diff(
              resource.data.get('documentos', {})
            ).affectedKeys().hasAny(['certificado']) &&
            (
              request.resource.data.get('disponible', false) != true ||
              (
                resource.data.estado == 'activo' &&
                resource.data.status == 'activo' &&
                resource.data.get('kyc', {}).get('aprobado', false) == true &&
                resource.data.get('suspendido', false) != true
              )
            )
          ) ||
          (userRole() == "admin_b2b" && resource.data.edificioId == userData().edificioId)
      );
    }

    // ======================================================
    // ♻️ COLECCIONES ESPEJO (MIGRACIÓN LEGACY)
    // ======================================================
    match /tecnicos/{userId} {
      allow read, write: if isAuth() && request.auth.uid == userId;
    }

    match /clientes/{userId} {
      allow read, write: if isAuth() && request.auth.uid == userId;
    }

    // ======================================================
    // 🛠️ ECOSISTEMA B2C (ON-DEMAND)
    // ======================================================
    match /services/{serviceId} {
      allow read: if isAuth() && (
        isAdmin() ||
        resource.data.cliente_id == request.auth.uid ||
        resource.data.get('tecnico_id', null) == request.auth.uid ||
        (resource.data.get('tipo', null) == 'mantenimiento' && isTecnico())
      );
      // B2C se crea sólo mediante createB2cService (Admin SDK). Se conserva el
      // contrato B2B independiente durante la transición.
      allow create: if isAuth() &&
        request.resource.data.get('metodo_pago', '') == 'b2b' &&
        request.resource.data.cliente_id == request.auth.uid &&
        userTipo() == 'B2B';
      allow update: if isAuth() && (
        isAdmin()
        ||
        (
          resource.data.cliente_id == request.auth.uid &&
          request.resource.data.cliente_id == resource.data.cliente_id &&
          request.resource.data.get('tecnico_id', null) == resource.data.get('tecnico_id', null) &&
          request.resource.data.get('destino', null) == resource.data.get('destino', null) &&
          serviceFinancialFieldsUnchanged() &&
          customerServiceUpdateIsSafe()
        )
        ||
        (
          resource.data.get('tecnico_id', null) == request.auth.uid &&
          isOperationalTechnician() &&
          request.resource.data.cliente_id == resource.data.cliente_id &&
          request.resource.data.tecnico_id == resource.data.tecnico_id &&
          request.resource.data.get('destino', null) == resource.data.get('destino', null) &&
          serviceFinancialFieldsUnchanged() &&
          technicianServiceTransitionIsSafe()
        )
        ||
        (
          resource.data.get('tipo', null) == 'mantenimiento' &&
          isTecnico() &&
          request.resource.data.get('tipo', null) == 'mantenimiento' &&
          serviceFinancialFieldsUnchanged()
        )
      );
    }

    // Proyección mínima de descubrimiento. El expediente completo permanece en services.
    match /service_marketplace/{serviceId} {
      allow read: if isOperationalTechnician() || isAdmin();
      allow write: if isAdmin();
    }

    match /platform_events/{eventId} {
      allow read: if isOperationalTechnician() || isAdmin();
      allow write: if isAdmin();
    }

    // El backend mantiene el candado de una sola misión activa por técnico.
    match /technician_active_services/{technicianId} {
      allow read: if isAdmin() || (isAuth() && request.auth.uid == technicianId);
      allow write: if isAdmin();
    }

    // ======================================================
    // 📢 SISTEMA DE NOTIFICACIONES (PUSH ENGINE)
    // ======================================================
    match /notificaciones_pendientes/{notiId} {
      // Permitimos crear si es Admin Maestro o personal autorizado (Megáfono activo para Operación B2B)
      // Incluye: Jorge (Admin/CEO), Jessica (Asistente), Mari Jo (Recepción), Luis y Laura (Seguridad)
      allow create: if isAuth() && (
        isAdmin() || 
        userRole() in [
          'admin_b2b', 
          'ceo', 
          'admin', 
          'asistente_admin', 
          'recepcion', 
          'seguridad', 
          'seguridad_interna',
          'seguridad_24_7'
        ]
      );
      
      // El acceso de lectura/borrado general queda reservado para el Admin o el Backend
      allow read: if isAuth();
      allow update, delete: if isAdmin();
    }
    
    // ======================================================
    // 📢 COMUNICADOS Y AVISOS OFICIALES (MEGÁFONO B2B)
    // ======================================================
    match /anuncios_b2b/{anuncioId} {
      // Cualquier usuario logueado en el edificio puede leer los anuncios
      allow read: if isAuth() && (
        isAdmin() || 
        userData().edificioId == resource.data.edificioId ||
        userRole() in ['admin_b2b', 'ceo', 'admin', 'asistente_admin', 'recepcion', 'seguridad', 'seguridad_interna', 'seguridad_24_7']
      );
      
      // Solo el equipo operativo del edificio puede crear anuncios
      allow create: if isAuth() && (
        isAdmin() || 
        userRole() in ['admin_b2b', 'ceo', 'admin', 'asistente_admin', 'recepcion', 'seguridad', 'seguridad_interna', 'seguridad_24_7']
      );
      
      // Edición y borrado reservados para administración y creadores (opcional, para mantener el blindaje)
      allow update, delete: if isAdmin() || (isAuth() && request.auth.uid == resource.data.autor_uid);
    }

    // Reglas para Alertas de Seguridad NOC
    match /alertas_seguridad/{document} {
      allow read, write: if isAuth();
    }
    
    // ======================================================
    // 🚀 MOTOR UXMAL 39 (B2B)
    // ======================================================
    match /servicios_b2b/{servicioId} {
      allow read: if isAuth() && (
        isAdmin() || 
        isTecnico() || 
        userRole() == 'admin_b2b' || 
        userTipo() == 'B2B'
      );
      
      allow create: if isAuth() && (
        isAdmin() || 
        isTecnico() || 
        userTipo() == 'B2B'
      );

      allow update: if isAuth() && (
        isAdmin() || 
        (isTecnico() && 
         userData().edificioId == resource.data.edificioId && 
         tienePaseCaseta()) ||
        (userTipo() == 'B2B' && 
         resource.data.edificioId == userData().edificioId)
      );
    }

    // ======================================================
    // 📋 RUTINAS PREVENTIVAS Y BITÁCORAS
    // ======================================================
    match /config_rutinas/{edificioId} {
      allow read: if isAuth() && (isTecnico() || isAdmin() || userTipo() == 'B2B');
      allow write: if isAdmin() || (isAuth() && userTipo() == 'B2B');
    }

    match /log_rutinas/{logId} {
      allow read: if isAuth() && (isTecnico() || isAdmin() || userTipo() == 'B2B');
      allow create: if isAuth() && 
                    isTecnico() && 
                    request.resource.data.tecnicoId == request.auth.uid && 
                    tienePaseCaseta();
    }

    match /bitacora_edificios/{bitId} {
      allow read, create: if isAuth() && 
                          (isTecnico() || isAdmin() || userTipo() == 'B2B');
    }

    // ======================================================
    // 💰 FINANZAS Y RASTREO
    // ======================================================
    match /transacciones/{txId} {
      allow read: if isAuth() && 
                  (resource.data.tecnico_id == request.auth.uid || isAdmin());
      allow create: if isAuth();
    }

    match /retiros/{retiroId} {
      allow read: if isAuth() && 
                  (resource.data.tecnico_id == request.auth.uid || isAdmin());
      allow create: if isAuth() && 
                    request.resource.data.tecnico_id == request.auth.uid;
    }

    match /rastreo/{userId} {
      allow read: if isAuth();
      allow write: if isAuth() && 
                   (request.auth.uid == userId || isAdmin());
    }

    // ======================================================
    // 🏢 ESTRUCTURA EMPRESARIAL B2B
    // ======================================================
    match /empresas_b2b/{empresaId} {
      allow read: if isAuth();

      match /areas/{areaId} {
        allow read, write: if isAuth();
      }

      // 🛠️ MÓDULO ACTUALIZADO: ACTIVOS (Catálogo Técnico Enriquecido)
      match /activos/{activoId} {
        allow read: if isAuth();
        
        // Escritura protegida para Administradores, validando la estructura del JSON nuevo
        allow create, update: if isAuth() && (
            isAdmin() || 
            (userRole() == 'admin_b2b' && userData().edificioId == empresaId)
        )
        // Validar que si mandan especificaciones técnicas, sea un objeto y los números sean números
        && (!('especificaciones_tecnicas' in request.resource.data) || request.resource.data.especificaciones_tecnicas is map)
        && (!('especificaciones_tecnicas' in request.resource.data) || !('potencia_hp' in request.resource.data.especificaciones_tecnicas) || request.resource.data.especificaciones_tecnicas.potencia_hp is number);

        match /bitacora/{bitId} {
          allow create, read: if isAuth();
        }
      }

      // 🛠️ NUEVO MÓDULO: MANTENIMIENTO (Logística Relacional Uxmal 39)
      match /mantenimiento/{tareaId} {
        allow read: if isAuth(); 
        
        // Creación reservada para Admins. Blindamos que no metan basura técnica aquí.
        allow create: if isAuth() && (
            isAdmin() || 
            (userRole() == 'admin_b2b' && userData().edificioId == empresaId)
        )
        && request.resource.data.keys().hasAll(['id_tarea', 'id_activo', 'sistema', 'frecuencia', 'estado'])
        && request.resource.data.estado in ['pendiente', 'en_proceso', 'completado']
        && !('potencia_hp' in request.resource.data)
        && !('voltaje' in request.resource.data);
                      
        // Actualización: Aquí entra el técnico. SOLO puede cambiar si es su edificio y tiene pase.
        allow update: if isAuth() && (
            isAdmin() || 
            (userRole() == 'admin_b2b' && userData().edificioId == empresaId) || 
            (isTecnico() && userData().edificioId == empresaId && tienePaseCaseta())
        )
        // Solo permitimos los estados válidos de la tarea
        && (!('estado' in request.resource.data) || request.resource.data.estado in ['pendiente', 'en_proceso', 'completado'])
        && !('potencia_hp' in request.resource.data);
      }
    }

    // ======================================================
    // 🧠 MOTOR NO-CODE: GESTIA TERMINAL (IA BUILDER)
    // ======================================================

    // ⚠️ IMPORTANTE:
    // - NO redefinimos userData()
    // - NO redefinimos userRole()
    // - Solo usamos las funciones globales ya existentes

    // ======================================================
    // 1. MÓDULOS (CONFIGURACIÓN)
    // ======================================================
    match /gestia_system_modules/{moduloId} {
      allow read: if isAuth();
      allow write: if isAdmin();
    }

    // ======================================================
    // 2. REGISTROS OPERATIVOS (CORE B2B)
    // ======================================================

    match /gestia_records/{tenantId} {

      // 🔥 acceso base (colección intermedia)
      allow read: if true;

      // 🔥 subcolección (AQUÍ ENTRA onSnapshot)
      match /{moduloId} {

        allow read: if true;

        // 🔥 documentos dentro del módulo
        match /{docId} {
          allow read, write: if true;
        }
      }
    }

    // ======================================================
    // 3. CONDOMINIOS (LISTA NEGRA / METADATA)
    // ======================================================
    match /condominios/{condoId} {

      function accesoCondominio() {
        let edificio = userTenant();
        return edificio != null && edificio == condoId;
      }

      allow read: if isAuth() && (
        isAdmin() ||
        accesoCondominio()
      );

      allow write: if isAdmin();
    }

    // ======================================================
    // CONFIGURACION GLOBAL
    // ======================================================
    match /configuracion/{docId} {
      allow read: if isAuth();
      allow write: if isAdmin();
    }

    // ======================================================
    // MODULO FLEET MANAGEMENT (FLOTILLA B2B)
    // ======================================================
    match /flotilla_b2b/{tenantId}/{document=**} {
      allow read, write: if isAuth(); 
    }

    // ======================================================
    // 📦 PACKAGES (DEBUG)
    // ======================================================

    match /packages/{tenantId}/items/{docId} {
      allow read, write: if true;
    }

    match /packages/{tenantId}/{docId} {
      allow read, write: if true;
    }

  }
}
