function b2cServiceExists(serviceId) {
  return exists(/databases/$(database)/documents/services/$(serviceId));
}

function b2cService(serviceId) {
  return get(/databases/$(database)/documents/services/$(serviceId)).data;
}

function b2cIsAssignedTechnician(serviceId) {
  return isAuth() &&
    b2cServiceExists(serviceId) &&
    b2cService(serviceId).tecnico_id == request.auth.uid;
}

function b2cIsOwningCustomer(serviceId) {
  return isAuth() &&
    b2cServiceExists(serviceId) &&
    b2cService(serviceId).cliente_id == request.auth.uid;
}

function b2cIsServiceParty(serviceId) {
  return isAdmin() ||
    b2cIsAssignedTechnician(serviceId) ||
    b2cIsOwningCustomer(serviceId);
}

// El orquestador técnico usa camelCase y objeto actor; el flujo cliente service-scoped
// usa snake_case. Estas funciones normalizan ambos contratos sin relajar identidad.
function b2cEventServiceId(data) {
  return "serviceId" in data ? data.serviceId : data.service_id;
}

function b2cEventType(data) {
  return "eventType" in data ? data.eventType : data.event_type;
}

function b2cEventActorUid(data) {
  return "actor" in data ? data.actor.uid : data.actor_uid;
}

function b2cEventActorRole(data) {
  return "actor" in data ? data.actor.role : data.actor_role;
}

function b2cEvidenceActorValid(serviceId, data) {
  let actorUid = b2cEventActorUid(data);
  let actorRole = b2cEventActorRole(data);
  return actorUid == request.auth.uid &&
    (
      (actorRole == "tecnico" && b2cIsAssignedTechnician(serviceId)) ||
      (actorRole == "cliente" && b2cIsOwningCustomer(serviceId))
    );
}

function b2cEvidenceEventAllowed(eventType) {
  return eventType in [
    "arrival",
    "arrival_fallback",
    "diagnostic_before",
    "work_before",
    "work_after",
    "customer_no_show",
    "customer_denied_access",
    "customer_no_show_video",
    "customer_denied_access_video",
    "service_completion",
    "customer_arrival_dispute"
  ];
}

function b2cTechnicianConsentValid(serviceId, expectedEvent) {
  return b2cIsAssignedTechnician(serviceId) &&
    request.resource.data.actor_uid == request.auth.uid &&
    request.resource.data.actor_role == "tecnico" &&
    request.resource.data.event_type == expectedEvent;
}

// IMPORTANTE: el match /services/{serviceId} existente NO cubre subcolecciones.
// Este segundo match con la misma ruta agrega únicamente permisos B2C anidados.
// No modifica todavía la regla legacy amplia de read/update del documento principal.
match /services/{serviceId} {

  match /evidence_events/{eventId} {
    allow read: if b2cIsServiceParty(serviceId);

    allow create: if b2cIsServiceParty(serviceId) &&
      b2cEventServiceId(request.resource.data) == serviceId &&
      b2cEvidenceActorValid(serviceId, request.resource.data) &&
      b2cEvidenceEventAllowed(b2cEventType(request.resource.data));

    allow update, delete: if isAdmin();
  }

  match /time_sync/{actorId} {
    allow read: if isAdmin() ||
      (b2cIsServiceParty(serviceId) && actorId == request.auth.uid);

    allow create, update: if b2cIsServiceParty(serviceId) &&
      actorId == request.auth.uid &&
      request.resource.data.actor_uid == request.auth.uid &&
      request.resource.data.actor_role in ["tecnico", "cliente"];

    allow delete: if isAdmin();
  }

  // Llegada, fallback, ausencia y negativa de acceso usan este consentimiento.
  match /capture_consents/{consentId} {
    allow read: if b2cIsServiceParty(serviceId);

    allow create, update: if b2cIsAssignedTechnician(serviceId) &&
      request.resource.data.actor_uid == request.auth.uid &&
      request.resource.data.actor_role == "tecnico" &&
      request.resource.data.event_type in [
        "arrival",
        "arrival_fallback",
        "customer_no_show",
        "customer_denied_access"
      ];

    allow delete: if isAdmin();
  }

  match /video_consents/{consentId} {
    allow read: if b2cIsServiceParty(serviceId);

    allow create, update: if b2cIsAssignedTechnician(serviceId) &&
      request.resource.data.actor_uid == request.auth.uid &&
      request.resource.data.actor_role == "tecnico" &&
      request.resource.data.event_type in [
        "customer_no_show_video",
        "customer_denied_access_video"
      ];

    allow delete: if isAdmin();
  }

  match /diagnostic_consents/{consentId} {
    allow read: if b2cIsServiceParty(serviceId);
    allow create, update: if b2cTechnicianConsentValid(
      serviceId,
      "diagnostic_before"
    );
    allow delete: if isAdmin();
  }

  match /work_start_consents/{consentId} {
    allow read: if b2cIsServiceParty(serviceId);
    allow create, update: if b2cTechnicianConsentValid(
      serviceId,
      "work_before"
    );
    allow delete: if isAdmin();
  }

  match /work_after_consents/{consentId} {
    allow read: if b2cIsServiceParty(serviceId);
    allow create, update: if b2cTechnicianConsentValid(
      serviceId,
      "work_after"
    );
    allow delete: if isAdmin();
  }

  // Compatibilidad temporal del guardia de cierre anterior.
  match /work_evidence_consents/{consentId} {
    allow read: if b2cIsServiceParty(serviceId);

    allow create, update: if b2cIsAssignedTechnician(serviceId) &&
      request.resource.data.actor_uid == request.auth.uid &&
      request.resource.data.actor_role == "tecnico" &&
      request.resource.data.event_type in ["work_before", "work_after"];

    allow delete: if isAdmin();
  }

  match /work_evidence_bindings/{bindingId} {
    allow read: if b2cIsServiceParty(serviceId);

    allow create, update: if b2cIsAssignedTechnician(serviceId) &&
      request.resource.data.technician_id == request.auth.uid &&
      request.resource.data.service_id == serviceId;

    allow delete: if isAdmin();
  }

  // Reserva SHA-256 del cliente confinada al mismo folio.
  match /customer_evidence_hashes/{hashId} {
    allow read: if b2cIsOwningCustomer(serviceId) || isAdmin();

    allow create: if b2cIsOwningCustomer(serviceId) &&
      request.resource.data.customer_id == request.auth.uid &&
      request.resource.data.scope == "service_only" &&
      request.resource.data.dedup_backend_pending == true &&
      request.resource.data.sha256 is string;

    allow update: if b2cIsOwningCustomer(serviceId) &&
      resource.data.customer_id == request.auth.uid &&
      request.resource.data.customer_id == resource.data.customer_id &&
      request.resource.data.sha256 == resource.data.sha256 &&
      request.resource.data.evidence_id == resource.data.evidence_id &&
      request.resource.data.scope == "service_only" &&
      request.resource.data.dedup_backend_pending == true;

    allow delete: if isAdmin();
  }
}

// Registro global del técnico. El cliente nunca obtiene acceso.
// get permite comprobar un SHA exacto conocido; list exige historial propio.
match /b2c_evidence_hashes/{hashId} {
  allow get: if isAdmin() || isTecnico();

  allow list: if isAdmin() ||
    (isTecnico() && resource.data.technicianId == request.auth.uid);

  allow create: if isTecnico() &&
    request.resource.data.technicianId == request.auth.uid &&
    b2cIsAssignedTechnician(request.resource.data.serviceId) &&
    request.resource.data.sha256 is string;

  allow update: if isTecnico() &&
    resource.data.technicianId == request.auth.uid &&
    request.resource.data.technicianId == resource.data.technicianId &&
    request.resource.data.serviceId == resource.data.serviceId &&
    request.resource.data.evidenceId == resource.data.evidenceId &&
    request.resource.data.sha256 == resource.data.sha256 &&
    b2cIsAssignedTechnician(resource.data.serviceId);

  allow delete: if isAdmin();
}

match /b2c_evidence_fingerprints/{fingerprintId} {
  allow get: if isAdmin() ||
    (isTecnico() && resource.data.technicianId == request.auth.uid);

  allow list: if isAdmin() ||
    (isTecnico() && resource.data.technicianId == request.auth.uid);

  allow create: if isTecnico() &&
    request.resource.data.technicianId == request.auth.uid &&
    b2cIsAssignedTechnician(request.resource.data.serviceId);

  allow update, delete: if isAdmin();
}

match /b2c_evidence_audit/{auditId} {
  allow read: if isAdmin();

  allow create: if isTecnico() &&
    request.resource.data.technicianId == request.auth.uid &&
    b2cIsAssignedTechnician(request.resource.data.serviceId);

  allow update, delete: if false;
}
